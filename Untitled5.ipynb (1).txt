        "    return value\n",

        "def _require_0_100(name: str, value: float) -> float:\n",
        "    value = _require_finite(name, value)\n",
        "    if not (0.0 <= value <= 100.0):\n",

        "    return value\n",

        "def calculate_score(oe: float, cv: float, vis: float) -> float:\n",
        "    return 0.6 * _require_0_100(\"OE\", oe) + 0.3 * _require_0_100(\"CV\", cv) + 0.1 * _require_0_100(\"VIS\", vis)\n",

        "    return 0.1 * _require_finite(\"SD\", sd) + 0.9 * _require_finite(\"CW\", cw)\n",
        "\n",
        "def calculate_liv(ri: float, sr: float, ar: float, ids: float) -> float:\n",

        "    if ids == 0.0:\n",
